    24→export function PlayerView({ initialRoom, onExit }: { initialRoom: string; onExit: () => void }) {
    25→  const socketRef = useRef(getGameSocket())
    26→  const connected = useSocketConnected()
    27→  const [state, setState] = useState<PlayerState | null>(null)
    28→  const [error, setError] = useState<string | null>(null)
    29→  const [busy, setBusy] = useState(false)
    30→
    31→  // ---- join form state (prefilled from storage at mount; client-only app) ----
    32→  const [init] = useState(() => {
    33→    try {
    34→      return {
    35→        name: localStorage.getItem(LS.name) ?? '',
    36→        code: localStorage.getItem(LS.code) ?? '',
    37→        id: localStorage.getItem(LS.id) ?? '',
    38→      }
    39→    } catch {
    40→      return { name: '', code: '', id: '' }
    41→    }
    42→  })
    43→  const [name, setName] = useState(init.name)
    44→  const [room, setRoom] = useState(initialRoom || init.code)
    45→
    46→  useEffect(() => {
    47→    const socket = socketRef.current
    48→    if (init.id && (initialRoom || init.code)) {
    49→      socket.emit('player:hello', { code: (initialRoom || init.code).toUpperCase(), playerId: init.id })
    50→    }
    51→
    52→    const onState = (s: PlayerState) => {
    53→      setState(s)
    54→      setError(null)
    55→      setBusy(false)
    56→      try {
    57→        localStorage.setItem(LS.id, s.you.id)
    58→        localStorage.setItem(LS.code, s.code)
    59→        localStorage.setItem(LS.name, s.you.name)
    60→      } catch {}
    61→    }
    62→    const onJoinError = (e: { message?: string }) => {
    63→      setBusy(false)
    64→      setError(e?.message ?? 'Could not join.')
    65→      try {
    66→        localStorage.removeItem(LS.id)
    67→      } catch {}
    68→    }
    69→    const onReset = () => {
    70→      setState(null)
    71→      setError('The host started a fresh session — rejoin with the new room code.')
    72→      try {
    73→        localStorage.removeItem(LS.id)
    74→      } catch {}
    75→    }
    76→
    77→    socket.on('player:state', onState)
    78→    socket.on('join:error', onJoinError)
    79→    socket.on('room:reset', onReset)
    80→
    81→    return () => {
    82→      socket.off('player:state', onState)
    83→      socket.off('join:error', onJoinError)
    84→      socket.off('room:reset', onReset)
    85→    }
    86→  }, [initialRoom, init])
    87→
    88→  const handleJoin = useCallback(() => {
    89→    if (!name.trim() || room.trim().length !== 4 || busy) return
    90→    setBusy(true)
    91→    socketRef.current.emit('player:hello', { code: room.toUpperCase(), name: name.trim() })
    92→  }, [name, room, busy])
    93→
    94→  const exit = useCallback(() => {
    95→    try {
    96→      localStorage.removeItem(LS.id)
    97→    } catch {}
    98→    onExit()
    99→  }, [onExit])
   100→
   101→  // ---------------- Join screen ----------------
   102→  if (!state) {
   103→    return (
   104→      <main className="mx-auto flex min-h-screen w-full max-w-md flex-col justify-center px-6 py-10">
   105→        <motion.div {...fade} className="space-y-6">
   106→          <div className="text-center">
   107→            <p className="text-[11px] font-bold uppercase tracking-[0.3em] text-amber-400">Two-Thirds Arena</p>
   108→            <h1 className="mt-2 text-3xl font-black tracking-tight">Join the class game</h1>
   109→          </div>
   110→
   111→          <div className="space-y-4 rounded-3xl border border-zinc-800 bg-zinc-900/50 p-6">
   112→            <div className="space-y-2">
   113→              <label htmlFor="nick" className="text-xs font-bold uppercase tracking-widest text-zinc-400">
   114→                Nickname
   115→              </label>
   116→              <Input
   117→                id="nick"
   118→                value={name}
   119→                onChange={(e) => setName(e.target.value.slice(0, 14))}
   120→                placeholder="e.g. Ananya"
   121→                autoComplete="off"
   122→                className="h-12 border-zinc-700 bg-zinc-950 text-base font-bold"
   123→                onKeyDown={(e) => e.key === 'Enter' && handleJoin()}
   124→              />
   125→            </div>
   126→            <div className="space-y-2">
   127→              <label htmlFor="room" className="text-xs font-bold uppercase tracking-widest text-zinc-400">
   128→                Room code (on the big screen)
   129→              </label>
   130→              <Input
   131→                id="room"
   132→                value={room}
   133→                onChange={(e) => setRoom(e.target.value.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 4))}
   134→                placeholder="ABCD"
   135→                className="h-12 border-zinc-700 bg-zinc-950 text-center text-2xl font-black tracking-[0.4em]"
   136→                autoComplete="off"
   137→                inputMode="text"
   138→                onKeyDown={(e) => e.key === 'Enter' && handleJoin()}
   139→              />
   140→            </div>
   141→
   142→            {error && (
   143→              <p className="rounded-lg border border-rose-500/30 bg-rose-500/10 px-3 py-2 text-sm font-semibold text-rose-300" role="alert">
   144→                {error}
   145→              </p>
   146→            )}
   147→
   148→            <Button
   149→              onClick={handleJoin}
   150→              disabled={!connected || !name.trim() || room.length !== 4 || busy}
   151→              className="h-14 w-full bg-amber-400 text-lg font-black text-zinc-950 hover:bg-amber-300"
   152→            >
   153→              {busy ? <Loader2 className="mr-2 h-5 w-5 animate-spin" aria-hidden="true" /> : null}
   154→              {connected ? 'ENTER THE ARENA' : 'CONNECTING…'}
   155→            </Button>
   156→            <button onClick={exit} className="flex w-full items-center justify-center gap-1 pt-1 text-xs font-semibold text-zinc-500 hover:text-zinc-300">
   157→              <ArrowLeft className="h-3 w-3" aria-hidden="true" /> back
   158→            </button>
   159→          </div>
   160→
   161→          <p className="text-center text-xs leading-relaxed text-zinc-600">
   162→            Keep this tab open — you&apos;ll play every round from here. If your WiFi blips, just reload: you&apos;ll rejoin automatically.
   163→          </p>
   164→        </motion.div>
   165→      </main>
   166→    )
   167→  }
   168→
   169→  // ---------------- In-game ----------------
   170→  return (
   171→    <main className="mx-auto flex min-h-screen w-full max-w-md flex-col px-5 pb-8 pt-4">
   172→      {/* header */}
   173→      <header className="flex items-center justify-between gap-2 rounded-2xl border border-zinc-800 bg-zinc-900/50 px-4 py-2.5">
   174→        <div className="flex items-center gap-2 overflow-hidden">
   175→          <span className="rounded-md bg-amber-400/15 px-2 py-1 text-[10px] font-black tracking-[0.2em] text-amber-300">
   176→            {state.code}
   177→          </span>
   178→          <span className="truncate text-sm font-bold text-zinc-200">{state.you.name}</span>
   179→        </div>
   180→        <div className="flex shrink-0 items-center gap-2">
   181→          <span className="inline-flex items-center gap-1 rounded-full bg-zinc-800 px-2.5 py-1 text-xs font-black tabular-nums text-zinc-100">
   182→            <TrendingUp className="h-3 w-3 text-amber-400" aria-hidden="true" />
   183→            {state.you.totalScore}
   184→          </span>
   185→          {connected ? (
   186→            <Wifi className="h-3.5 w-3.5 text-emerald-400" aria-label="Connected" />
   187→          ) : (
   188→            <WifiOff className="h-3.5 w-3.5 animate-pulse text-rose-400" aria-label="Reconnecting" />
   189→          )}
   190→        </div>
   191→      </header>
   192→
   193→      {!connected && (
   194→        <p className="mt-3 rounded-xl border border-rose-500/30 bg-rose-500/10 px-3 py-2 text-center text-xs font-bold text-rose-300" role="alert">
   195→          Connection lost — reconnecting…
   196→        </p>
   197→      )}
   198→
   199→      <AnimatePresence mode="wait">
   200→        <motion.div key={state.phase} {...fade} transition={{ duration: 0.25 }} className="mt-5 flex flex-1 flex-col">
   201→          <PhaseScreen state={state} socketRef={socketRef} />
   202→        </motion.div>
   203→      </AnimatePresence>
   204→    </main>
   205→  )
   206→}

   210→function PhaseScreen({ state, socketRef }: { state: PlayerState; socketRef: React.RefObject<ReturnType<typeof getGameSocket>> }) {
   211→  switch (state.phase) {
   212→    case 'lobby':
   213→      return <WaitingRoom state={state} />
   214→    case 'howto':
   215→      return <HowTo state={state} />
   216→    case 'r1_open':
   217→    case 'r2_open':
   218→    case 'r3_open':
   219→      return <GuessPanel key={state.round} state={state} socketRef={socketRef} />
   220→    case 'r1_reveal':
   221→    case 'r2_reveal':
   222→    case 'r3_reveal':
   223→    case 'r1_debrief':
   224→    case 'r2_debrief':
   225→    case 'r3_debrief':
   226→      return <RoundResultCard state={state} />
   227→    case 'leaderboard':
   228→    case 'finale_intro':
   229→      return <LeaderboardCard state={state} />
   230→    case 'finale_vote':
   231→      return <FinaleVote state={state} socketRef={socketRef} />
   232→    case 'finale_reveal':
   233→    case 'finale_debrief':
   234→      return <FinaleResultCard state={state} />
   235→    case 'wrap':
   236→      return <WrapCard state={state} />
   237→    default:
   238→      return <WaitingRoom state={state} />

   295→function GuessPanel({
   296→  state,
   297→  socketRef,
   298→}: {
   299→  state: PlayerState
   300→  socketRef: React.RefObject<ReturnType<typeof getGameSocket>>
   301→}) {
   302→  const [draft, setDraft] = useState<number>(50)
   303→  const [typed, setTyped] = useState<string>('50')
   304→  const [locked, setLocked] = useState(false)
   305→  const round = state.round
   306→
   307→  const lockIn = useCallback(() => {
   308→    const v = Math.min(100, Math.max(0, Number(typed)))
   309→    if (!Number.isFinite(v)) return
   310→    setDraft(v)
   311→    socketRef.current.emit('player:guess', { guess: v })
   312→    setLocked(true)
   313→  }, [typed, socketRef])
   314→
   315→  const setBoth = (v: number) => {
   316→    const clamped = Math.min(100, Math.max(0, Math.round(v * 100) / 100))
   317→    setDraft(clamped)
   318→    setTyped(String(clamped))
   319→  }
   320→
   321→  if (!state.you.canPlayNow) {
   322→    return (
   323→      <div className="flex flex-1 flex-col items-center justify-center gap-4 text-center">
   324→        <Eye className="h-14 w-14 text-zinc-600" aria-hidden="true" />
   325→        <h2 className="text-xl font-black">Round {round} is live</h2>
   326→        <p className="max-w-xs text-sm leading-relaxed text-zinc-400">
   327→          You joined mid-round — sit tight and watch the big screen. You&apos;re in from the next round.
   328→        </p>
   329→      </div>
   330→    )
   331→  }
   332→
   333→  const submitted = state.you.yourGuess !== null
   334→
   335→  return (
   336→    <div className="flex flex-1 flex-col justify-center gap-6">
   337→      {state.endsAt && <TimerBar endsAt={state.endsAt} />}
   338→
   339→      <div className="text-center">
   340→        <p className="text-[11px] font-black uppercase tracking-[0.3em] text-amber-400">Round {round} of 3</p>
   341→        <h2 className="mt-1 text-2xl font-black leading-tight">
   342→          Pick a number
   343→          <span className="block text-base font-bold text-zinc-400">0 – 100</span>
   344→        </h2>
   345→      </div>
   346→
   347→      {/* big number */}
   348→      <div className="rounded-3xl border border-amber-500/25 bg-gradient-to-b from-amber-400/10 to-transparent p-6 text-center">
   349→        <p className="text-7xl font-black tabular-nums text-amber-300">{typed === '' ? '—' : typed}</p>
   350→        <p className="mt-1 text-[10px] font-bold uppercase tracking-[0.25em] text-zinc-500">your guess</p>
   351→      </div>
   352→
   353→      <div className="space-y-4">
   354→        <Slider
   355→          value={[draft]}
   356→          min={0}
   357→          max={100}
   358→          step={1}
   359→          onValueChange={(v) => setBoth(v[0])}
   360→          disabled={locked && submitted}
   361→          aria-label="Your guess from 0 to 100"
   362→          className="[&_span:first-child]:h-3 [&_.bg-primary]:bg-amber-400"
   363→        />
   364→        <div className="flex items-center gap-3">
   365→          <Input
   366→            value={typed}
   367→            onChange={(e) => {
   368→              const raw = e.target.value
   369→              if (raw === '' || /^(\d{0,3}(\.\d{0,2})?)$/.test(raw)) setTyped(raw)
   370→            }}
   371→            onBlur={() => {
   372→              const v = Number(typed)
   373→              setBoth(Number.isFinite(v) ? v : 50)
   374→            }}
   375→            disabled={locked && submitted}
   376→            inputMode="decimal"
   377→            aria-label="Type your guess"
   378→            className="h-12 flex-1 border-zinc-700 bg-zinc-950 text-center text-xl font-black tabular-nums"
   379→          />
   380→          {locked && submitted ? (
   381→            <Button
   382→              variant="outline"
   383→              onClick={() => setLocked(false)}
   384→              className="h-12 gap-2 border-zinc-700 bg-zinc-900 font-bold text-zinc-200"
   385→            >
   386→              <Pencil className="h-4 w-4" aria-hidden="true" /> Change
   387→            </Button>
   388→          ) : (
   389→            <Button
   390→              onClick={lockIn}
   391→              disabled={typed === '' || !Number.isFinite(Number(typed))}
   392→              className="h-12 gap-2 bg-amber-400 px-6 text-base font-black text-zinc-950 hover:bg-amber-300"
   393→            >
   394→              <Lock className="h-4 w-4" aria-hidden="true" /> Lock in
   395→            </Button>
   396→          )}
   397→        </div>
   398→      </div>
   399→
   400→      {submitted && (
   401→        <div className="rounded-2xl border border-emerald-500/25 bg-emerald-500/10 px-4 py-3 text-center">
   402→          <p className="text-sm font-bold text-emerald-300">
   403→            Locked: {fmtNum(state.you.yourGuess ?? 0)} · {state.submittedCount}/{state.eligibleCount} in
   404→          </p>
   405→          <p className="mt-0.5 text-[11px] text-emerald-400/70">You can still change it until time runs out.</p>
   406→        </div>
   407→      )}
   408→    </div>
   409→  )
   410→}

   412→function TimerBar({ endsAt }: { endsAt: number }) {
   413→  const [now, setNow] = useState(() => Date.now())
   414→  useEffect(() => {
   415→    const t = setInterval(() => setNow(Date.now()), 250)
   416→    return () => clearInterval(t)
   417→  }, [])
   418→  const remaining = Math.max(0, endsAt - now)
   419→  const seconds = Math.ceil(remaining / 1000)
   420→  return (
   421→    <div className="flex items-center justify-between rounded-xl border border-zinc-800 bg-zinc-900/60 px-4 py-2.5" role="timer">
   422→      <span className="inline-flex items-center gap-1.5 text-xs font-bold text-zinc-400">
   423→        <Timer className="h-3.5 w-3.5" aria-hidden="true" /> time left
   424→      </span>
   425→      <span className={`text-lg font-black tabular-nums ${seconds <= 10 ? 'text-rose-400' : 'text-zinc-100'}`}>{seconds}s</span>
   426→    </div>
   427→  )
   428→}
